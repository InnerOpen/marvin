✅ Post-deployment verification checklist:
   1. Check pod status:
      oc get pods -l app.kubernetes.io/name={{ include "marvin.name" . }}

   2. Verify health endpoint (the API is internal by default, so port-forward the Service):
{{- if eq .Values.mode "split" }}
      oc port-forward svc/{{ include "marvin.fullname" . }}-backend 8080:{{ .Values.service.port }}
{{- else }}
      oc port-forward svc/{{ include "marvin.fullname" . }} 8080:{{ .Values.service.port }}
{{- end }}
      curl -v http://localhost:8080/healthz

   3. Check scheduler state file:
      oc exec $POD -- cat {{ .Values.config.dataDir }}/scheduler_state.json

   4. Verify no tokens in logs:
      oc logs $POD | grep -i "token.*:" | head -20
      (Should NOT show plaintext JWTs)

   5. Reach the web UI:
{{- if and .Values.route.enabled .Values.route.ui.enabled }}
      open "https://$UI"
{{- else if eq .Values.mode "split" }}
      oc port-forward svc/{{ include "marvin.fullname" . }}-frontend 4322:{{ .Values.service.frontendPort }}
      open http://localhost:4322
{{- else }}
      oc port-forward svc/{{ include "marvin.fullname" . }} 4322:{{ .Values.service.frontendPort }}
      open http://localhost:4322
{{- end }}
